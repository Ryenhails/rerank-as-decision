% the mark: a rounded tile with four probability bars and a target, drawn at width #1
\newcommand{\vlmjevmark}[1]{%
  \begin{tikzpicture}[x=#1/360, y=#1/360, line cap=round]
    \draw[jevline2, line width=#1/180, fill=white, rounded corners=#1*88/360] (2,2) rectangle (358,358);
    \shade[shading=jevgrad, rounded corners=#1*18/360] (64,252) rectangle (224,288);
    \draw[jevrose, line width=#1/45] (272,270) circle[radius=28];
    \fill[jevrose] (272,270) circle[radius=13];
    \fill[jevred, rounded corners=#1*18/360] (64,192) rectangle (168,228);
    \fill[jevyellow, rounded corners=#1*18/360] (64,132) rectangle (128,168);
    \fill[jevgreen, rounded corners=#1*18/360] (64,72) rectangle (100,108);
  \end{tikzpicture}}
% the target icon of the cover, at height #1
\newcommand{\jevtarget}[1]{%
  \tikz[baseline=-0.5ex]{\draw[jevrose, line width=#1*0.12] (0,0) circle[radius=#1*0.38]; \fill[jevrose] (0,0) circle[radius=#1*0.18];}}
